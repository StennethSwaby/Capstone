slam_toolbox_bringup

Bring-up script: Run TurtleBot3 + SLAM Toolbox + Teleop (Real Robot)

# 0) Network & environment (do on BOTH robot and laptop)
export ROS_DOMAIN_ID=30            # pick any small integer; must match on both
source /opt/ros/humble/setup.bash
export TURTLEBOT3_MODEL=burger     # or waffle_pi

# 1) Robot side (on the Raspberry Pi)
# SSH into the robot, then run:
source /opt/ros/humble/setup.bash
export ROS_DOMAIN_ID=30
export TURTLEBOT3_MODEL=burger
ros2 launch turtlebot3_bringup robot.launch.py
# Keep this terminal running. It brings up LiDAR (/scan), odometry (/odom), TF, motors.

# 2) Laptop side: start SLAM Toolbox (do NOT use sim time on real robot)
source /opt/ros/humble/setup.bash
export ROS_DOMAIN_ID=30
export TURTLEBOT3_MODEL=burger
# Option A: use your bringup package (if you made a real-robot launch)
# ros2 launch slam_toolbox_bringup slam_toolbox_real.launch.py
# Option B: use the stock online SLAM launch (works out-of-the-box)
ros2 launch slam_toolbox online_async_launch.py use_sim_time:=false

# 3) Laptop side: teleop to move the robot and build the map
source /opt/ros/humble/setup.bash
export ROS_DOMAIN_ID=30
ros2 run teleop_twist_keyboard teleop_twist_keyboard
# Use arrows/WASD. If your robot is namespaced, remap cmd_vel, e.g.:
# ros2 run teleop_twist_keyboard teleop_twist_keyboard --ros-args -r /cmd_vel:=/tb3/cmd_vel

# 4) (Optional) Visualize in RViz (on laptop)
rviz2
# In RViz: Fixed Frame=map; add Map(/map), LaserScan(/scan), TF, RobotModel.

# 5) Save the map (on laptop) when you’re satisfied
ros2 run nav2_map_server map_saver_cli -f ~/tb3_map
# Produces ~/tb3_map.yaml and ~/tb3_map.pgm

# Quick health checks (on laptop)
ros2 topic list | grep -E '^/map$|/scan|/odom|/tf'
ros2 node list | grep slam_toolbox
ros2 topic hz /map
ros2 topic echo /scan --once
ros2 run tf2_tools view_frames    # creates frames.pdf (map->odom->base_link)

# Common fixes
# - No /scan or /odom: check robot bringup (Step 1) and DOMAIN_ID match.
# - Map not updating: move the robot; ensure use_sim_time:=false on real robot.
# - Teleop not moving: check /cmd_vel topic or remap for namespaces.
# - QoS issues (scan drops): keep SLAM Toolbox default QoS or match LaserScan QoS BEST_EFFORT.
